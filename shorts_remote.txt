function ensureState(){
  fs.mkdirSync(DATA_DIR,{recursive:true});
  if(!fs.existsSync(STATE_FILE)) fs.writeFileSync(STATE_FILE, JSON.stringify({version:1,items:[],topicOffsets:{},exhausted:{},updatedAt:0},null,2));
  if(!fs.existsSync(BLOOM_FILE)) fs.writeFileSync(BLOOM_FILE, Buffer.alloc(BLOOM_BYTES));
}
function loadState(){ensureState();try{return JSON.parse(fs.readFileSync(STATE_FILE,'utf8'));}catch{return {version:1,items:[],topicOffsets:{},exhausted:{},updatedAt:0};}}
function saveState(state){state.items=state.items.slice(-CACHE_MAX_ITEMS);state.updatedAt=Date.now();fs.writeFileSync(STATE_FILE,JSON.stringify(state));}
function loadBloom(){ensureState();let b=fs.readFileSync(BLOOM_FILE);if(b.length!==BLOOM_BYTES){b=Buffer.alloc(BLOOM_BYTES);}return b;}
function saveBloom(b){fs.writeFileSync(BLOOM_FILE,b);}
function hashes(value){const out=[];for(let i=0;i<BLOOM_HASHES;i++){const h=crypto.createHash('sha256').update(i+':'+value).digest();out.push(h.readUInt32BE(0)%BLOOM_BITS);}return out;}
function bloomHas(b,value){return hashes(value).every(bit=>(b[bit>>3]&(1<<(bit&7)))!==0);}
function bloomAdd(b,value){for(const bit of hashes(value))b[bit>>3]|=1<<(bit&7);}
function idForUrl(url){return 'rshort-'+crypto.createHash('sha256').update(url).digest('hex').slice(0,24);}
function b64url(value){return Buffer.from(String(value)).toString('base64url');}
function allowedLicense(license){const s=String(license||'').replace(/<[^>]+>/g,'').toLowerCase();return ['cc0','public domain','cc by','cc by-sa'].some(x=>s.includes(x));}
async function searchCommons(query,offset=0,limit=50){
  const qs=new URLSearchParams({action:'query',generator:'search',gsrsearch:query,gsrnamespace:'6',gsrlimit:String(Math.min(limit,50)),gsroffset:String(offset),prop:'imageinfo',iiprop:'url|mime|extmetadata',format:'json',origin:'*'});
  const r=await fetch('https://commons.wikimedia.org/w/api.php?'+qs.toString(),{headers:{'User-Agent':DEFAULT_USER_AGENT},signal:AbortSignal.timeout(12000)});
  if(!r.ok) throw new Error('Commons API '+r.status);
  const j=await r.json();
  const items=Object.values(j.query?.pages||{}).map(page=>{
    const ii=page.imageinfo?.[0]||{},md=ii.extmetadata||{};
    return {url:ii.url||'',mime:ii.mime||'',title:String(page.title||'').replace(/^File:/,''),license:md.LicenseShortName?.value||md.License?.value||'',sourcePage:'https://commons.wikimedia.org/wiki/'+encodeURIComponent(String(page.title||'')),sourceCreator:md.Artist?.value||md.Credit?.value||''};
  }).filter(x=>/^video\/(mp4|webm|ogg)$/i.test(x.mime)&&x.url&&allowedLicense(x.license));
  return {items,nextOffset:j.continue?.gsroffset ?? null};
}

function topicEntries(category){
  const wanted=String(category||'').toUpperCase();
  const source=wanted&&wanted!=='ALL'?TOPIC_SEEDS.filter(([c])=>c===wanted):TOPIC_SEEDS;
  return source.flatMap(([cat,topics])=>topics.map(topic=>({category:cat,topic}))); 
}
function makeItem(raw,category,topic,index){
  const id=idForUrl(raw.url);
  return {id,category:rawCategory(category,raw,topic),title:raw.title.replace(/\.[^.]+$/,'').slice(0,120)||topic,description:`Real-world ${topic} footage from Wikimedia Commons.`,tags:[category,topic].join(' ').split(/\s+/).filter(Boolean).slice(0,8),duration:0,sourceFile:raw.title,sourcePage:raw.sourcePage,sourceCreator:raw.sourceCreator,license:raw.license,audio:true,mediaSource:'Wikimedia Commons',remoteUrl:raw.url,remoteKey:b64url(raw.url),creatorIndex:index%12,createdAt:new Date().toISOString()};
}
function rawCategory(category,raw,topic){return category&&category!=='ALL'?category:String(raw.title+' '+topic).toUpperCase().split(/[^A-Z0-9]+/).filter(Boolean)[0]||'GENERAL';}

async function discover({needed=12,category='ALL'}={}){
  const state=loadState();const bloom=loadBloom();const current=state.items.filter(x=>!category||category==='ALL'||x.category===category);
  if(current.length>=needed)return current.slice(0,needed);
  const topics=topicEntries(category);let guard=0;
  while(state.items.filter(x=>!category||category==='ALL'||x.category===category).length<needed && guard<topics.length*3){
    const batch=topics.slice((guard%Math.max(1,topics.length)),(guard%Math.max(1,topics.length))+5); if(!batch.length)break;
    const results=await Promise.all(batch.map(async t=>{
      const key=t.category+'|'+t.topic; if(state.exhausted[key])return [t,null];
      const offset=Number(state.topicOffsets[key]||0);
      try{return [t,await searchCommons(t.topic+' video',offset,50)]}catch(err){return [t,{error:err.message,items:[],nextOffset:null}]}
    }));
    for(const [t,result] of results){
      if(!result)continue;
      const key=t.category+'|'+t.topic;
      if(result.nextOffset===null)state.exhausted[key]=true; else state.topicOffsets[key]=result.nextOffset||0;
      for(const raw of result.items){
        if(state.items.some(x=>x.remoteUrl===raw.url))continue;
        if(bloomHas(bloom,raw.url))continue;
        bloomAdd(bloom,raw.url);
        state.items.push(makeItem(raw,t.category,t.topic,state.items.length));
        if(state.items.filter(x=>!category||category==='ALL'||x.category===category).length>=needed)break;
      }
    }
    guard+=5;
    if(guard>topics.length*3)break;
  }
  saveState(state);saveBloom(bloom);
  return state.items.filter(x=>!category||category==='ALL'||x.category===category).slice(0,needed);
}

async function getPage({page=1,limit=12,category='ALL'}={}){
  const state=loadState();
  const seed=state.items.filter(x=>!category||category==='ALL'||x.category===category);
  const needed=Math.min(SHORTS_CAPACITY,page*limit);
  if(seed.length<needed){try{await discover({needed,category});}catch{} }
  const final=loadState().items.filter(x=>!category||category==='ALL'||x.category===category);
  const start=(page-1)*limit;
  return {items:final.slice(start,start+limit),hasMore:page*limit<SHORTS_CAPACITY && (final.length>start+limit || final.length<SHORTS_CAPACITY),available:final.length,capacity:SHORTS_CAPACITY};
}

function decodeRemoteKey(key){try{return Buffer.from(String(key),'base64url').toString('utf8');}catch{return '';}}
function isAllowedRemoteUrl(url){try{const u=new URL(url);return u.protocol==='https:' && (u.hostname==='commons.wikimedia.org'||u.hostname.endsWith('.wikimedia.org'));}catch{return false;}}
